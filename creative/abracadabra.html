This minimalist Python program deconstructs the famous magic incantation by recombining its syllables, to create new incantations. Read aloud at your own risk.

import random
selections = ["a", "bra", "ca", "da", "bra"]
def generate_word():
    return ''.join(random.choices(selections, k=5))
for _ in range(100):
    print(generate_word())
